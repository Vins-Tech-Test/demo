import numpy as np

def binary_step(x):
    return 1 if x >= 0 else 0

def linear(x):
    return x

def sigmoid(x):
    return 1 / (1 + np.exp(-x))

def tanh(x):
    return np.tanh(x)

def relu(x):
    return max(0, x)

def leaky_relu(x):
    return x if x > 0 else 0.01 * x

def elu(x):
    return x if x > 0 else np.exp(x) - 1

def softmax(x):
    e = np.exp(x - np.max(x))
    return e / np.sum(e)


x = -2

print("Binary Step:", binary_step(x))
print("Linear:", linear(x))
print("Sigmoid:", sigmoid(x))
print("Tanh:", tanh(x))
print("ReLU:", relu(x))
print("Leaky ReLU:", leaky_relu(x))
print("ELU:", elu(x))

x = np.array([1, 2, 3])
print("Softmax:", softmax(x))
